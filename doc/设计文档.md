### 3.4 抓取调度

使用 GitHub Actions 定时执行抓取任务：

```yaml
# .github/workflows/fetch.yml
name: Fetch Free Games
on:
  schedule:
    - cron: '0 */6 * * *'  # 每 6 小时执行一次
  workflow_dispatch:  # 支持手动触发

jobs:
  fetch:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
      - run: npm ci
      - run: npm run fetch:all
      - name: Commit and push
        run: |
          git config user.name "GitHub Actions"
          git config user.email "actions@github.com"
          git add data/
          git commit -m "chore: update free games data" || exit 0
          git push
```

### 4.1 PlatformFilter 组件

```vue
<!-- components/PlatformFilter.vue -->
<script setup lang="ts">
import platforms from '~/data/platforms.json'

const emit = defineEmits(['change'])
const selected = ref<string[]>(platforms.map(p => p.id))

function toggle(platformId: string) {
  const index = selected.value.indexOf(platformId)
  if (index > -1) {
    selected.value.splice(index, 1)
  } else {
    selected.value.push(platformId)
  }
  emit('change', selected.value)
}
</script>

<template>
  <div class="flex flex-wrap gap-2">
    <button
      v-for="p in platforms"
      :key="p.id"
      @click="toggle(p.id)"
      class="px-3 py-1.5 rounded-full text-sm font-medium transition"
      :class="selected.includes(p.id) ? 'bg-brand text-white' : 'bg-slate-100 text-slate-600 hover:bg-slate-200'"
    >
      {{ p.name }}
    </button>
  </div>
</template>
```

## 5. 业务逻辑设计

### 5.1 useGames 增强

```typescript
// composables/useGames.ts
import gamesData from '~/data/games.json'
import historyData from '~/data/history.json'
import platformsData from '~/data/platforms.json'

export interface Game {
  id: string
  title: string
  description: string
  image: string
  originalPrice: string
  startDate: string
  endDate: string
  slug: string
  url: string
  isCurrent: boolean
  platform: string
  seller: string
  fetchedAt: string
}

export interface HistoryGame {
  id: string
  title: string
  image: string
  originalPrice: string
  startDate: string
  endDate: string
  slug: string
  url: string
  platform: string
}

export interface Platform {
  id: string
  name: string
  icon: string
  color: string
}

export const useGames = () => {
  const games = gamesData as Game[]
  const history = historyData as HistoryGame[]
  const platforms = platformsData.platforms as Platform[]

  return {
    current: games.filter(g => g.isCurrent),
    upcoming: games.filter(g => !g.isCurrent),
    history: [...history].sort((a, b) =>
      new Date(b.endDate).getTime() - new Date(a.endDate).getTime()
    ),
    all: games,
    platforms,
    
    // 新增：按平台筛选
    filterByPlatform: (platformIds: string[]) => 
      games.filter(g => platformIds.includes(g.platform)),
    
    // 新增：搜索
    search: (query: string) => {
      const q = query.toLowerCase()
      return games.filter(g => 
        g.title.toLowerCase().includes(q) ||
        g.description?.toLowerCase().includes(q)
      )
    },
    
    findBySlug: (slug: string): Game | undefined =>
      games.find(g => g.slug === slug),
  }
}
```

### 5.2 useFilter 组合式函数

```typescript
// composables/useFilter.ts
import { useGames } from './useGames'

export const useFilter = () => {
  const { all, platforms } = useGames()
  
  const selectedPlatforms = ref<string[]>(platforms.map(p => p.id))
  const searchQuery = ref('')
  const status = ref<'all' | 'current' | 'upcoming'>('all')

  const filteredGames = computed(() => {
    let result = [...all]
    
    // 平台筛选
    if (selectedPlatforms.value.length > 0) {
      result = result.filter(g => selectedPlatforms.value.includes(g.platform))
    }
    
    // 状态筛选
    if (status.value === 'current') {
      result = result.filter(g => g.isCurrent)
    } else if (status.value === 'upcoming') {
      result = result.filter(g => !g.isCurrent)
    }
    
    // 搜索筛选
    if (searchQuery.value) {
      const q = searchQuery.value.toLowerCase()
      result = result.filter(g => 
        g.title.toLowerCase().includes(q) ||
        g.description?.toLowerCase().includes(q)
      )
    }
    
    return result
  })

  return {
    selectedPlatforms,
    searchQuery,
    status,
    filteredGames,
  }
}
```
